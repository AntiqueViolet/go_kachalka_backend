# go_kachalka backend — часть 2: БД, сервисы, HTTP API. План реализации

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Работающий сервер авторизации: схема Postgres, сервисы регистрации/входа/refresh/сброса пароля/профиля, HTTP API по контракту OpenAPI, сборка приложения и бинарники `api` и `keygen`.

**Architecture:** `store` (sqlc поверх pgx) → `otp` + `codes` (коды и их доставка) → `auth` и `account` (бизнес-логика, возвращают `*apperr.Error`) → `httpapi` (сгенерированный strict server + middleware) → `app` (сборка зависимостей) → `cmd/api`. Сервисы тестируются на настоящем Postgres (testcontainers), HTTP-слой — через `httptest`.

**Tech Stack:** pgx/v5, sqlc 1.31.1 (через Docker), goose v3, testcontainers-go, oapi-codegen v2.8.0 (strict server, `go tool`), net/http.

**Спецификация:** `docs/superpowers/specs/2026-10-22-auth-backend-design.md`
**Части плана:** 1 — фундамент (должна быть выполнена) → 2 — этот файл → 3 — комплект поставки.

```sql
INSERT INTO users (id, phone, password_hash, email, created_at, updated_at)
VALUES (@id, @phone, @password_hash, @email, @now::timestamptz, @now::timestamptz)
RETURNING *;

-- name: GetUserByID :one
SELECT * FROM users WHERE id = $1;

-- name: GetUserByPhone :one
SELECT * FROM users WHERE phone = $1;

-- name: GetUserByVerifiedEmail :one
SELECT * FROM users WHERE email = $1 AND email_verified_at IS NOT NULL;

-- name: UpdatePasswordHash :exec
UPDATE users SET password_hash = @password_hash, updated_at = @now::timestamptz WHERE id = @id;

-- name: SetEmail :one
UPDATE users SET email = @email, email_verified_at = NULL, updated_at = @now::timestamptz
WHERE id = @id
RETURNING *;

-- name: MarkEmailVerified :one
UPDATE users SET email_verified_at = @now::timestamptz, updated_at = @now::timestamptz
WHERE id = @id
RETURNING *;

-- name: DeleteUser :exec
DELETE FROM users WHERE id = $1;

-- name: LockUser :one
-- Locks the user row until the transaction ends and returns the current password hash. Logins
-- (session creation) serialise with password changes and resets through this lock.
SELECT password_hash FROM users WHERE id = $1 FOR UPDATE;

-- name: RehashPassword :execrows
-- Upgrades a hash made with outdated parameters, unless the password changed in the meantime.
UPDATE users SET password_hash = @new_hash, updated_at = @now::timestamptz
WHERE id = @id AND password_hash = @old_hash;
```

`internal/store/queries/sessions.sql`:

<!-- file: internal/store/queries/sessions.sql -->
```sql
-- name: CreateSession :one
INSERT INTO sessions (id, user_id, device_name, generation, created_at, last_used_at, expires_at)
VALUES (@id, @user_id, @device_name, 1, @now::timestamptz, @now::timestamptz, @expires_at)
RETURNING *;

-- name: GetSession :one
SELECT * FROM sessions WHERE id = $1;

-- name: GetSessionForUpdate :one
SELECT * FROM sessions WHERE id = $1 FOR UPDATE;

-- name: RotateSession :one
UPDATE sessions
SET generation = generation + 1, rotated_at = @now::timestamptz, last_used_at = @now::timestamptz, expires_at = @expires_at
WHERE id = @id
RETURNING *;

-- name: TouchSession :exec
-- GREATEST: a slow request must not move last_used_at backwards (the session limit orders by it).
UPDATE sessions SET last_used_at = GREATEST(last_used_at, @now::timestamptz) WHERE id = @id;

-- name: RevokeSession :exec
UPDATE sessions SET revoked_at = @now::timestamptz, revoke_reason = @reason::text
WHERE id = @id AND revoked_at IS NULL;

-- name: RevokeUserSessions :exec
UPDATE sessions SET revoked_at = @now::timestamptz, revoke_reason = @reason::text
WHERE user_id = @user_id AND revoked_at IS NULL;

-- name: RevokeUserSessionsExcept :exec
UPDATE sessions SET revoked_at = @now::timestamptz, revoke_reason = @reason::text
WHERE user_id = @user_id AND id <> @keep_id AND revoked_at IS NULL;

-- name: RevokeExcessSessions :exec
-- Keeps the most recently used live sessions of the user (up to keep) and revokes the rest.
UPDATE sessions SET revoked_at = @now::timestamptz, revoke_reason = 'session_limit'
WHERE id IN (
    SELECT s.id FROM sessions s
    WHERE s.user_id = @user_id AND s.revoked_at IS NULL AND s.expires_at > @now::timestamptz
    ORDER BY s.last_used_at DESC, s.id DESC
    OFFSET @keep::int
);

-- name: DeleteStaleSessions :execrows
DELETE FROM sessions
WHERE (revoked_at IS NOT NULL AND revoked_at < @before::timestamptz) OR expires_at < @before::timestamptz;
```

```sql
-- name: CreateCode :one
INSERT INTO verification_codes (id, purpose, target, user_id, code_hash, expires_at, created_at)
VALUES (@id, @purpose, @target, @user_id, @code_hash, @expires_at, @now::timestamptz)
RETURNING *;

-- name: GetCode :one
SELECT * FROM verification_codes WHERE id = $1;

-- name: LatestActiveUserCode :one
SELECT * FROM verification_codes
WHERE purpose = @purpose AND user_id = @user_id::uuid AND consumed_at IS NULL
ORDER BY created_at DESC
LIMIT 1;

-- name: InvalidateTargetCodes :exec
UPDATE verification_codes SET consumed_at = @now::timestamptz
WHERE purpose = @purpose AND target = @target AND user_id IS NULL AND consumed_at IS NULL;

-- name: InvalidateUserCodes :exec
UPDATE verification_codes SET consumed_at = @now::timestamptz
WHERE purpose = @purpose AND user_id = @user_id::uuid AND consumed_at IS NULL;

-- name: CodeSendStats :one
-- How many codes were sent to the address within the window, plus the oldest and latest send time.
SELECT count(*)::bigint AS sent,
       coalesce(min(created_at), @since)::timestamptz AS oldest,
       coalesce(max(created_at), @since)::timestamptz AS latest
FROM verification_codes
WHERE target = @target AND created_at > @since;

-- name: ClaimCodeAttempt :one
-- Burns one attempt before the code is compared. No row means the code is consumed, expired or
-- out of attempts; the check runs in the same statement, so parallel guesses can't exceed the limit.
UPDATE verification_codes SET attempts = attempts + 1
WHERE id = @id AND consumed_at IS NULL AND expires_at > @now::timestamptz AND attempts < @max_attempts::int
RETURNING attempts;

-- name: ConsumeCode :execrows
UPDATE verification_codes SET consumed_at = @now::timestamptz WHERE id = @id AND consumed_at IS NULL;

-- name: DeleteOldCodes :execrows
DELETE FROM verification_codes WHERE created_at < @before;

-- name: LockCodeTarget :exec
-- Serialises code issuing per address until the transaction ends, so parallel requests see each
-- other's codes when they check the send limits.
SELECT pg_advisory_xact_lock(hashtextextended(@target::text, 0));

-- name: DeleteCode :exec
-- Removes a code that could not be delivered, so it does not count against the send limits.
DELETE FROM verification_codes WHERE id = $1;
```

```go
// Several instances may start at once against an empty database (a deploy that scales up):
// the migration lock makes one of them migrate and the others wait, so all of them start.
func TestConcurrentMigrate(t *testing.T) {
	admin := storetest.Pool(t)
	ctx := context.Background()
	cfg, err := pgxpool.ParseConfig(storetest.DSN(t))
	if err != nil {
		t.Fatal(err)
	}
	cfg.ConnConfig.Database = "migrate_race"
	drop := func() {
		if _, err := admin.Exec(ctx, "DROP DATABASE IF EXISTS migrate_race WITH (FORCE)"); err != nil {
			t.Fatal(err)
		}
	}
	t.Cleanup(drop)
	for round := range 3 {
		drop()
		if _, err := admin.Exec(ctx, "CREATE DATABASE migrate_race"); err != nil {
			t.Fatal(err)
		}
		errs := make(chan error, 6)
		var wg sync.WaitGroup
		for range cap(errs) {
			wg.Add(1)
			go func() {
				defer wg.Done()
				pool, err := pgxpool.NewWithConfig(ctx, cfg)
				if err != nil {
					errs <- err
					return
				}
				defer pool.Close()
				errs <- store.Migrate(ctx, pool)
			}()
		}
		wg.Wait()
		close(errs)
		for err := range errs {
			if err != nil {
				t.Fatalf("round %d: %v", round+1, err)
			}
		}
	}
}
```

```go
// Package store is the Postgres data layer. Query code is generated by sqlc from
// queries/*.sql into *.sql.go, db.go and models.go; regenerate with the command in README.
package store

import (
	"context"
	"errors"

	"github.com/jackc/pgx/v5"
	"github.com/jackc/pgx/v5/pgconn"
	"github.com/jackc/pgx/v5/pgxpool"
)

// Constraint names referenced by the services.
const (
	ConstraintUserPhone     = "users_phone_key"
	ConstraintVerifiedEmail = "users_verified_email_uq"
)

// Store runs queries on the pool (autocommit) or inside a transaction via InTx.
type Store struct {
	*Queries
	pool *pgxpool.Pool
}

func NewStore(pool *pgxpool.Pool) *Store {
	return &Store{Queries: New(pool), pool: pool}
}

// InTx runs fn in a transaction and commits if fn returns nil.
func (s *Store) InTx(ctx context.Context, fn func(q *Queries) error) error {
	return pgx.BeginFunc(ctx, s.pool, func(tx pgx.Tx) error {
		return fn(s.WithTx(tx))
	})
}

func (s *Store) Ping(ctx context.Context) error { return s.pool.Ping(ctx) }

func IsNotFound(err error) bool { return errors.Is(err, pgx.ErrNoRows) }

// IsUniqueViolation reports a unique violation on the named constraint or unique index.
func IsUniqueViolation(err error, constraint string) bool {
	var pgErr *pgconn.PgError
	return errors.As(err, &pgErr) && pgErr.Code == "23505" && pgErr.ConstraintName == constraint
}
```

```go
package store

import (
	"context"
	"fmt"

	"github.com/jackc/pgx/v5/pgxpool"
	"github.com/jackc/pgx/v5/stdlib"
	"github.com/pressly/goose/v3"
	"github.com/pressly/goose/v3/lock"

	"github.com/AntiqueViolet/go_kachalka_backend/migrations"
)

// Migrate applies pending migrations embedded in the binary.
func Migrate(ctx context.Context, pool *pgxpool.Pool) error {
	db := stdlib.OpenDBFromPool(pool)
	defer func() { _ = db.Close() }() // releases the wrapper only; the pool stays open
	// The session lock (pg_advisory_lock) lets several instances start at once: one migrates,
	// the others wait (polling every second, up to 5 minutes) and then find nothing to do.
	locker, err := lock.NewPostgresSessionLocker(lock.WithLockTimeout(1, 300))
	if err != nil {
		return fmt.Errorf("migrate: %w", err)
	}
	p, err := goose.NewProvider(goose.DialectPostgres, db, migrations.FS, goose.WithSessionLocker(locker))
	if err != nil {
		return fmt.Errorf("migrate: %w", err)
	}
	if _, err := p.Up(ctx); err != nil {
		return fmt.Errorf("migrate: %w", err)
	}
	return nil
}
```

```go
// parallel queries really runs in parallel instead of queueing behind connection setup (race
// tests rely on it).
func Warm(t *testing.T, p *pgxpool.Pool, n int) {
	t.Helper()
	conns := make([]*pgxpool.Conn, min(n, int(p.Config().MaxConns)))
	var wg sync.WaitGroup
	errs := make(chan error, len(conns))
	for i := range conns {
		wg.Add(1)
		go func() {
			defer wg.Done()
			c, err := p.Acquire(context.Background())
			conns[i] = c
			errs <- err
		}()
	}
	wg.Wait()
	close(errs)
	for _, c := range conns {
		if c != nil {
			c.Release()
		}
	}
	for err := range errs {
		if err != nil {
			t.Fatalf("storetest: warm: %v", err)
		}
	}
```

```go
package otp

import (
	"context"
	"time"

	"github.com/google/uuid"

	"github.com/AntiqueViolet/go_kachalka_backend/internal/apperr"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/clock"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/store"
)

const sendWindow = time.Hour

type Limits struct {
	Cooldown         time.Duration // minimum gap between codes to one address
	PerTargetPerHour int
}

// Service issues and checks codes stored in verification_codes.
type Service struct {
	key    []byte
	limits Limits
	clock  clock.Clock
}

func NewService(key []byte, limits Limits, c clock.Clock) *Service {
	return &Service{key: key, limits: limits, clock: c}
}

func (s *Service) Cooldown() time.Duration { return s.limits.Cooldown }

type Issued struct {
	ID   uuid.UUID
	Code string
}

// Issue creates a code after checking send limits for target. Earlier active codes with the
// same purpose are invalidated: per user when userID is set, otherwise per target.
// q must be a transaction: Issue locks target until it ends, so parallel requests can't all pass
// the limit check or leave several active codes. The caller sends Code after the commit.
func (s *Service) Issue(ctx context.Context, q *store.Queries, purpose Purpose, target string, userID *uuid.UUID) (Issued, error) {
	if err := q.LockCodeTarget(ctx, target); err != nil {
		return Issued{}, err
	}
	now := s.clock.Now()
	if err := s.checkLimits(ctx, q, target, now); err != nil {
		return Issued{}, err
	}
	var err error
	if userID != nil {
		err = q.InvalidateUserCodes(ctx, store.InvalidateUserCodesParams{Now: now, Purpose: string(purpose), UserID: *userID})
	} else {
		err = q.InvalidateTargetCodes(ctx, store.InvalidateTargetCodesParams{Now: now, Purpose: string(purpose), Target: target})
	}
	if err != nil {
		return Issued{}, err
	}
	code, err := Generate()
	if err != nil {
		return Issued{}, err
	}
	id, err := uuid.NewV7()
	if err != nil {
		return Issued{}, err
	}
	if _, err := q.CreateCode(ctx, store.CreateCodeParams{
		ID: id, Purpose: string(purpose), Target: target, UserID: userID,
		CodeHash: Hash(s.key, id, code), ExpiresAt: now.Add(TTL), Now: now,
	}); err != nil {
		return Issued{}, err
	}
	return Issued{ID: id, Code: code}, nil
}

func (s *Service) checkLimits(ctx context.Context, q *store.Queries, target string, now time.Time) error {
	st, err := q.CodeSendStats(ctx, store.CodeSendStatsParams{Target: target, Since: now.Add(-sendWindow)})
	if err != nil {
		return err
	}
	if st.Sent == 0 {
		return nil
	}
	if wait := st.Latest.Add(s.limits.Cooldown).Sub(now); wait > 0 {
		return apperr.RateLimited(wait)
	}
	if st.Sent >= int64(s.limits.PerTargetPerHour) {
		return apperr.RateLimited(st.Oldest.Add(sendWindow).Sub(now))
	}
	return nil
}

// Check verifies input against c. Every call burns one attempt first, atomically in the database,
// so parallel requests can't try more than MaxAttempts codes. Pass the autocommit store queries,
// not a transaction, so a later rollback can't return the attempt. Check does not consume the
// code: the caller calls Consume inside its own transaction.
func (s *Service) Check(ctx context.Context, q *store.Queries, c store.VerificationCode, input string) error {
	attempts, err := q.ClaimCodeAttempt(ctx, store.ClaimCodeAttemptParams{ID: c.ID, Now: s.clock.Now(), MaxAttempts: MaxAttempts})
	if store.IsNotFound(err) {
		return apperr.ErrCodeExpired
	}
	if err != nil {
		return err
	}
	if Matches(s.key, c.ID, input, c.CodeHash) {
		return nil
	}
	return apperr.InvalidCode(max(MaxAttempts-int(attempts), 0))
}

// Consume marks the code used. It fails with ErrCodeExpired if a concurrent request won.
func (s *Service) Consume(ctx context.Context, q *store.Queries, id uuid.UUID) error {
	n, err := q.ConsumeCode(ctx, store.ConsumeCodeParams{ID: id, Now: s.clock.Now()})
	if err != nil {
		return err
	}
	if n == 0 {
		return apperr.ErrCodeExpired
	}
	return nil
}
```

```go
	"github.com/AntiqueViolet/go_kachalka_backend/internal/otp"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/store"
)

// backgroundTimeout bounds a background send: database work plus one SMTP delivery (10 s).
// Shutdown waits for background sends, so it also keeps the stop within the compose grace period.
const backgroundTimeout = 12 * time.Second

type Deps struct {
	Store *store.Store
	OTP   *otp.Service
	SMS   notify.SMSSender
	Email notify.EmailSender
	Texts notify.Texts
	Log   *slog.Logger
}

type Sender struct {
	d  Deps
	bg sync.WaitGroup // background sends, see SendEmailCodeInBackground
}

func NewSender(d Deps) *Sender { return &Sender{d: d} }

// Cooldown is how long a client must wait before asking for another code to the same address.
func (s *Sender) Cooldown() time.Duration { return s.d.OTP.Cooldown() }

// SendPhoneCode issues a registration code for phone and sends it by SMS.
// Delivery failure is reported as ErrUnavailable: without the SMS the user cannot continue.
func (s *Sender) SendPhoneCode(ctx context.Context, phone string) (uuid.UUID, error) {
	is, err := s.issue(ctx, otp.PurposePhoneRegister, phone, nil)
	if err != nil {
		return uuid.Nil, err
	}
	text, err := s.d.Texts.PhoneCode(is.Code)
	if err != nil {
		s.discard(ctx, is.ID)
		return uuid.Nil, err
	}
	if err := s.d.SMS.SendSMS(ctx, phone, text); err != nil {
		s.discard(ctx, is.ID)
		return uuid.Nil, s.failed(ctx, "sms", err)
	}
	return is.ID, nil
}

// SendEmailCode issues an email_verify or password_reset code for the user and emails it.
func (s *Sender) SendEmailCode(ctx context.Context, purpose otp.Purpose, userID uuid.UUID, email string) error {
	is, err := s.issue(ctx, purpose, email, &userID)
	if err != nil {
		return err
	}
	if err := s.DeliverEmailCode(ctx, purpose, email, is.Code); err != nil {
		s.discard(ctx, is.ID)
		return err
	}
	return nil
}

// SendEmailCodeInBackground runs SendEmailCode after the caller has answered, so neither the
// answer nor its timing tells whether the address belongs to an account. Errors are only logged.
func (s *Sender) SendEmailCodeInBackground(ctx context.Context, purpose otp.Purpose, userID uuid.UUID, email string) {
	ctx = context.WithoutCancel(ctx)
	s.bg.Add(1)
	go func() {
		defer s.bg.Done()
		ctx, cancel := context.WithTimeout(ctx, backgroundTimeout)
		defer cancel()
		if err := s.SendEmailCode(ctx, purpose, userID, email); err != nil {
			s.d.Log.WarnContext(ctx, "code not sent", "purpose", purpose, "err", err)
		}
	}()
}

// Wait blocks until background sends have finished (graceful shutdown, tests).
func (s *Sender) Wait() { s.bg.Wait() }

// DeliverEmailCode emails a code that the caller issued inside its own transaction.
func (s *Sender) DeliverEmailCode(ctx context.Context, purpose otp.Purpose, email, code string) error {
	render := s.d.Texts.EmailVerification
	if purpose == otp.PurposePasswordReset {
		render = s.d.Texts.PasswordReset
	}
	subject, body, err := render(code)
	if err != nil {
		return err
	}
	if err := s.d.Email.SendEmail(ctx, email, subject, body); err != nil {
		return s.failed(ctx, "email", err)
	}
	return nil
}

// issue creates the code in a transaction of its own, so the per-address lock that otp.Issue
// takes covers both the send-limit check and the insert.
func (s *Sender) issue(ctx context.Context, purpose otp.Purpose, target string, userID *uuid.UUID) (otp.Issued, error) {
	var is otp.Issued
	err := s.d.Store.InTx(ctx, func(q *store.Queries) error {
		var err error
		is, err = s.d.OTP.Issue(ctx, q, purpose, target, userID)
		return err
	})
	return is, err
}

// discard deletes a code that was never delivered, so a provider failure does not use up the
// address's send limits.
func (s *Sender) discard(ctx context.Context, id uuid.UUID) {
	if err := s.d.Store.DeleteCode(context.WithoutCancel(ctx), id); err != nil {
		s.d.Log.WarnContext(ctx, "undelivered code not deleted", "err", err)
	}
}

func (s *Sender) failed(ctx context.Context, channel string, err error) error {
	s.d.Log.ErrorContext(ctx, "code delivery failed", "channel", channel, "err", err)
	return apperr.ErrUnavailable
}
```

```go
const (
	// MaxSessions is the number of live sessions a user may have; older ones are revoked.
	MaxSessions   = 10
	maxDeviceName = 100
)

type Deps struct {
	Store         *store.Store
	Hasher        *password.Hasher
	Signer        *token.Signer
	Refresh       *token.RefreshCodec
	OTP           *otp.Service
	Codes         *codes.Sender
	Clock         clock.Clock
	Log           *slog.Logger
	LoginFailures *ratelimit.Limiter // failed logins per phone
	RefreshTTL    time.Duration
	RefreshGrace  time.Duration
}

type Service struct{ d Deps }

func New(d Deps) *Service { return &Service{d: d} }

type Tokens struct {
	Access           string
	AccessExpiresIn  time.Duration
	Refresh          string
	RefreshExpiresIn time.Duration
}

type Result struct {
	User   store.User
	Tokens Tokens
}

// newSession creates a session and trims the user's live sessions to MaxSessions. The caller
// holds the user row lock (Login takes it, Register has just inserted the row), so concurrent
// logins trim one after another and the limit stays exact.
func (s *Service) newSession(ctx context.Context, q *store.Queries, userID uuid.UUID, device string) (store.Session, error) {
	now := s.d.Clock.Now()
	id, err := uuid.NewV7()
	if err != nil {
		return store.Session{}, err
	}
	sess, err := q.CreateSession(ctx, store.CreateSessionParams{
		ID: id, UserID: userID, DeviceName: device, Now: now, ExpiresAt: now.Add(s.d.RefreshTTL),
	})
	if err != nil {
		return store.Session{}, err
	}
	err = q.RevokeExcessSessions(ctx, store.RevokeExcessSessionsParams{UserID: userID, Now: now, Keep: MaxSessions})
	return sess, err
}

func (s *Service) tokensFor(sess store.Session) (Tokens, error) {
	access, err := s.d.Signer.IssueAccess(sess.UserID, sess.ID)
	if err != nil {
		return Tokens{}, err
	}
	return Tokens{
		Access:           access,
		AccessExpiresIn:  s.d.Signer.AccessTTL(),
		Refresh:          s.d.Refresh.Encode(sess.ID, sess.Generation),
		RefreshExpiresIn: sess.ExpiresAt.Sub(s.d.Clock.Now()),
	}, nil
}

func normalizeDevice(name string) (string, *apperr.FieldError) {
	name = strings.TrimSpace(name)
	switch {
	case name == "":
		return "unknown", nil
	case utf8.RuneCountInString(name) > maxDeviceName:
		return "", &apperr.FieldError{Field: "device_name", Code: apperr.FieldTooLong}
	}
	return name, nil
}

func invalidPhone() error {
	return apperr.Validation(apperr.FieldError{Field: "phone", Code: apperr.FieldInvalidPhone})
}
```

```go
	ExpiresIn      time.Duration
	ResendAfter    time.Duration
}

// StartRegistration sends an SMS code to a phone number that is not registered yet.
func (s *Service) StartRegistration(ctx context.Context, rawPhone string) (Started, error) {
	phone, err := contact.NormalizePhone(rawPhone)
	if err != nil {
		return Started{}, invalidPhone()
	}
	switch _, err := s.d.Store.GetUserByPhone(ctx, phone); {
	case err == nil:
		return Started{}, apperr.ErrPhoneTaken
	case !store.IsNotFound(err):
		return Started{}, err
	}
	id, err := s.d.Codes.SendPhoneCode(ctx, phone)
	if err != nil {
		return Started{}, err
	}
	return Started{VerificationID: id, ExpiresIn: otp.TTL, ResendAfter: s.d.Codes.Cooldown()}, nil
}

type RegistrationToken struct {
	Token     string
	ExpiresIn time.Duration
}

// VerifyPhone checks the SMS code and returns a short-lived token proving phone ownership.
func (s *Service) VerifyPhone(ctx context.Context, verificationID uuid.UUID, code string) (RegistrationToken, error) {
	c, err := s.d.Store.GetCode(ctx, verificationID)
	if store.IsNotFound(err) || (err == nil && c.Purpose != string(otp.PurposePhoneRegister)) {
		return RegistrationToken{}, apperr.ErrCodeExpired
	}
	if err != nil {
		return RegistrationToken{}, err
	}
	if err := s.d.OTP.Check(ctx, s.d.Store.Queries, c, code); err != nil {
		return RegistrationToken{}, err
	}
	if err := s.d.OTP.Consume(ctx, s.d.Store.Queries, c.ID); err != nil {
		return RegistrationToken{}, err
	}
	tok, err := s.d.Signer.IssueRegistration(c.Target)
	if err != nil {
		return RegistrationToken{}, err
	}
	return RegistrationToken{Token: tok, ExpiresIn: s.d.Signer.RegistrationTTL()}, nil
}

type RegisterInput struct {
	RegistrationToken string
	Password          string
	Email             string
	DeviceName        string
}

// Register creates the account and its first session, then emails a verification code
// (best effort: the user can request a resend).
func (s *Service) Register(ctx context.Context, in RegisterInput) (Result, error) {
	phone, err := s.d.Signer.ParseRegistration(in.RegistrationToken)
	if err != nil {
		return Result{}, apperr.ErrInvalidRegistrationToken
	}
	var fields []apperr.FieldError
	email, err := contact.NormalizeEmail(in.Email)
	if err != nil {
		fields = append(fields, apperr.FieldError{Field: "email", Code: apperr.FieldInvalidEmail})
	}
	if fe := password.FieldError("password", in.Password); fe != nil {
		fields = append(fields, *fe)
	}
	device, fe := normalizeDevice(in.DeviceName)
	if fe != nil {
		fields = append(fields, *fe)
	}
	if len(fields) > 0 {
		return Result{}, apperr.Validation(fields...)
	}

	hash, err := s.d.Hasher.Hash(ctx, in.Password)
	if err != nil {
		return Result{}, err
	}
	userID, err := uuid.NewV7()
	if err != nil {
		return Result{}, err
	}
	var res Result
	err = s.d.Store.InTx(ctx, func(q *store.Queries) error {
		u, err := q.CreateUser(ctx, store.CreateUserParams{
			ID: userID, Phone: phone, PasswordHash: hash, Email: email, Now: s.d.Clock.Now(),
		})
		if store.IsUniqueViolation(err, store.ConstraintUserPhone) {
			return apperr.ErrPhoneTaken
		}
		if err != nil {
			return err
		}
		sess, err := s.newSession(ctx, q, u.ID, device)
		if err != nil {
			return err
		}
		res.User = u
		res.Tokens, err = s.tokensFor(sess)
		return err
	})
	if err != nil {
		return Result{}, err
	}
	if err := s.d.Codes.SendEmailCode(ctx, otp.PurposeEmailVerify, res.User.ID, res.User.Email); err != nil {
		s.d.Log.WarnContext(ctx, "email verification code not sent after registration", "err", err)
	}
	return res, nil
}
```

```go
package auth

import (
	"context"

	"github.com/AntiqueViolet/go_kachalka_backend/internal/apperr"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/contact"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/store"
)

type LoginInput struct {
	Phone      string
	Password   string
	DeviceName string
}

// Login checks phone and password and opens a new session.
// An unknown phone and a wrong password are indistinguishable, including by timing.
func (s *Service) Login(ctx context.Context, in LoginInput) (Result, error) {
	phone, err := contact.NormalizePhone(in.Phone)
	if err != nil {
		return Result{}, invalidPhone()
	}
	device, fe := normalizeDevice(in.DeviceName)
	if fe != nil {
		return Result{}, apperr.Validation(*fe)
	}
	// The attempt is reserved before the password is checked, so parallel guesses can't all slip
	// past the limit; anything but a wrong password gives it back, so only failures count.
	refund, allowed, wait := s.d.LoginFailures.Acquire(phone)
	if !allowed {
		return Result{}, apperr.RateLimited(wait)
	}
	failed := false
	defer func() {
		if !failed {
			refund()
		}
	}()

	u, err := s.d.Store.GetUserByPhone(ctx, phone)
	if store.IsNotFound(err) {
		if err := s.d.Hasher.VerifyDummy(ctx, in.Password); err != nil {
			return Result{}, err
		}
		failed = true
		return Result{}, apperr.ErrInvalidCredentials
	}
	if err != nil {
		return Result{}, err
	}
	ok, rehash, err := s.d.Hasher.Verify(ctx, in.Password, u.PasswordHash)
	if err != nil {
		return Result{}, err
	}
	if !ok {
		failed = true
		return Result{}, apperr.ErrInvalidCredentials
	}
	res := Result{User: u}
	err = s.d.Store.InTx(ctx, func(q *store.Queries) error {
		// The password may have been reset or changed while it was being checked. The lock makes
		// the reset wait for this session (and revoke it) or makes this login see the new hash;
		// it also serialises the session-limit trim of parallel logins.
		hash, err := q.LockUser(ctx, u.ID)
		if store.IsNotFound(err) {
			return apperr.ErrInvalidCredentials
		}
		if err != nil {
			return err
		}
		if hash != u.PasswordHash {
			// Changed since it was checked: a reset or password change (the password no longer
			// matches) or a parallel login's rehash (it still does). Rare, so checked again here.
			ok, _, err := s.d.Hasher.Verify(ctx, in.Password, hash)
			if err != nil {
				return err
			}
			if !ok {
				return apperr.ErrInvalidCredentials
			}
		}
		sess, err := s.newSession(ctx, q, u.ID, device)
		if err != nil {
			return err
		}
		res.Tokens, err = s.tokensFor(sess)
		return err
	})
	if err != nil {
		return Result{}, err
	}
	if rehash {
		s.rehash(ctx, u, in.Password)
	}
	return res, nil
}

// rehash upgrades a hash made with outdated parameters; failure only costs another try next login.
// It leaves the hash alone if the password changed in the meantime.
func (s *Service) rehash(ctx context.Context, u store.User, pw string) {
	hash, err := s.d.Hasher.Hash(ctx, pw)
	if err == nil {
		_, err = s.d.Store.RehashPassword(ctx, store.RehashPasswordParams{ID: u.ID, OldHash: u.PasswordHash, NewHash: hash, Now: s.d.Clock.Now()})
	}
	if err != nil {
		s.d.Log.WarnContext(ctx, "password rehash failed", "err", err)
	}
}
```

```go
package auth

import (
	"context"

	"github.com/AntiqueViolet/go_kachalka_backend/internal/apperr"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/store"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/token"
)

// Refresh exchanges a refresh token for a new token pair (spec section 5.3).
func (s *Service) Refresh(ctx context.Context, raw string) (Tokens, error) {
	sid, gen, err := s.d.Refresh.Decode(raw)
	if err != nil {
		return Tokens{}, apperr.ErrInvalidRefreshToken
	}
	var out Tokens
	reused := false
	err = s.d.Store.InTx(ctx, func(q *store.Queries) error {
		sess, err := q.GetSessionForUpdate(ctx, sid)
		if store.IsNotFound(err) {
			return apperr.ErrInvalidRefreshToken
		}
		if err != nil {
			return err
		}
		now := s.d.Clock.Now()
		if sess.RevokedAt != nil || !now.Before(sess.ExpiresAt) {
			return apperr.ErrInvalidRefreshToken
		}
		switch token.Decide(gen, sess.Generation, sess.RotatedAt, now, s.d.RefreshGrace) {
		case token.DecisionRotate:
			sess, err = q.RotateSession(ctx, store.RotateSessionParams{ID: sid, Now: now, ExpiresAt: now.Add(s.d.RefreshTTL)})
			if err != nil {
				return err
			}
		case token.DecisionReplay:
			if err := q.TouchSession(ctx, store.TouchSessionParams{ID: sid, Now: now}); err != nil {
				return err
			}
		case token.DecisionReuse:
			// Commit the revocation, then report the reuse outside the transaction.
			reused = true
			return q.RevokeSession(ctx, store.RevokeSessionParams{ID: sid, Now: now, Reason: "refresh_reuse"})
		default:
			return apperr.ErrInvalidRefreshToken
		}
		out, err = s.tokensFor(sess)
		return err
	})
	if err != nil {
		return Tokens{}, err
	}
	if reused {
		s.d.Log.WarnContext(ctx, "refresh token reuse detected, session revoked", "session_id", sid)
		return Tokens{}, apperr.ErrRefreshTokenReused
	}
	return out, nil
}

// Logout revokes the current session, or all sessions of the user. It is idempotent.
func (s *Service) Logout(ctx context.Context, a token.Access, allDevices bool) error {
	now := s.d.Clock.Now()
	if allDevices {
		return s.d.Store.RevokeUserSessions(ctx, store.RevokeUserSessionsParams{UserID: a.UserID, Now: now, Reason: "logout_all"})
	}
	return s.d.Store.RevokeSession(ctx, store.RevokeSessionParams{ID: a.SessionID, Now: now, Reason: "logout"})
}
```

```go
package auth

import (
	"context"
	"errors"
	"net/http"
	"time"

	"github.com/AntiqueViolet/go_kachalka_backend/internal/apperr"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/contact"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/otp"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/password"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/store"
)

// errResetCode is the one answer to a wrong, expired or used code and to an unknown address:
// distinct answers (or attempts_left) would tell whether the address belongs to an account.
var errResetCode = apperr.New(http.StatusUnprocessableEntity, "invalid_code", "Invalid code")

// ForgotPassword emails a reset code if the address is verified on some account. The answer
// never depends on whether it is, nor on send limits or delivery problems; the email goes out
// in the background so that the response time does not tell either.
func (s *Service) ForgotPassword(ctx context.Context, rawEmail string) (resendAfter time.Duration, err error) {
	email, err := contact.NormalizeEmail(rawEmail)
	if err != nil {
		return 0, apperr.Validation(apperr.FieldError{Field: "email", Code: apperr.FieldInvalidEmail})
	}
	resendAfter = s.d.Codes.Cooldown()
	u, err := s.d.Store.GetUserByVerifiedEmail(ctx, email)
	if store.IsNotFound(err) {
		return resendAfter, nil
	}
	if err != nil {
		return 0, err
	}
	s.d.Codes.SendEmailCodeInBackground(ctx, otp.PurposePasswordReset, u.ID, u.Email)
	return resendAfter, nil
}

type ResetInput struct {
	Email       string
	Code        string
	NewPassword string
}

// ResetPassword sets a new password and revokes every session of the user.
func (s *Service) ResetPassword(ctx context.Context, in ResetInput) error {
	var fields []apperr.FieldError
	email, err := contact.NormalizeEmail(in.Email)
	if err != nil {
		fields = append(fields, apperr.FieldError{Field: "email", Code: apperr.FieldInvalidEmail})
	}
	if fe := password.FieldError("new_password", in.NewPassword); fe != nil {
		fields = append(fields, *fe)
	}
	if len(fields) > 0 {
		return apperr.Validation(fields...)
	}

	u, err := s.d.Store.GetUserByVerifiedEmail(ctx, email)
	if store.IsNotFound(err) {
		return errResetCode
	}
	if err != nil {
		return err
	}
	c, err := s.d.Store.LatestActiveUserCode(ctx, store.LatestActiveUserCodeParams{Purpose: string(otp.PurposePasswordReset), UserID: u.ID})
	if store.IsNotFound(err) || (err == nil && c.Target != u.Email) { // a code sent to a previous address
		return errResetCode
	}
	if err != nil {
		return err
	}
	if err := s.d.OTP.Check(ctx, s.d.Store.Queries, c, in.Code); err != nil {
		return resetCodeError(err)
	}
	hash, err := s.d.Hasher.Hash(ctx, in.NewPassword)
	if err != nil {
		return err
	}
	now := s.d.Clock.Now()
	return s.d.Store.InTx(ctx, func(q *store.Queries) error {
		if err := s.d.OTP.Consume(ctx, q, c.ID); err != nil {
			return resetCodeError(err)
		}
		if err := q.UpdatePasswordHash(ctx, store.UpdatePasswordHashParams{ID: u.ID, PasswordHash: hash, Now: now}); err != nil {
			return err
		}
		return q.RevokeUserSessions(ctx, store.RevokeUserSessionsParams{UserID: u.ID, Now: now, Reason: "password_reset"})
	})
}

// resetCodeError hides why a code was refused, see errResetCode.
func resetCodeError(err error) error {
	var ae *apperr.Error
	if errors.As(err, &ae) && (ae.Code == "invalid_code" || ae.Code == apperr.ErrCodeExpired.Code) {
		return errResetCode
	}
	return err
}
```

```go
// Package account implements operations on the signed-in user's profile.
package account

import (
	"context"
	"log/slog"
	"time"

	"github.com/AntiqueViolet/go_kachalka_backend/internal/apperr"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/clock"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/codes"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/otp"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/password"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/ratelimit"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/store"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/token"
)

type Deps struct {
	Store  *store.Store
	Hasher *password.Hasher
	OTP    *otp.Service
	Codes  *codes.Sender
	Clock  clock.Clock
	Log    *slog.Logger
	// PasswordFailures limits wrong current-password attempts per user (password change, email
	// change, account deletion); nil disables the limit.
	PasswordFailures *ratelimit.Limiter
}

type Service struct{ d Deps }

func New(d Deps) *Service { return &Service{d: d} }

// CodeSent tells the client how long the code lives and when it may ask for another.
type CodeSent struct {
	ExpiresIn   time.Duration
	ResendAfter time.Duration
}

func (s *Service) codeSent() CodeSent {
	return CodeSent{ExpiresIn: otp.TTL, ResendAfter: s.d.Codes.Cooldown()}
}

func (s *Service) Me(ctx context.Context, a token.Access) (store.User, error) {
	u, err := s.d.Store.GetUserByID(ctx, a.UserID)
	if store.IsNotFound(err) {
		return store.User{}, apperr.ErrUnauthorized
	}
	return u, err
}

// activeUser is Me plus a database check that the token's session is still live.
// Sensitive operations use it instead of trusting a possibly revoked 15-minute access token.
func (s *Service) activeUser(ctx context.Context, a token.Access) (store.User, error) {
	sess, err := s.d.Store.GetSession(ctx, a.SessionID)
	if store.IsNotFound(err) {
		return store.User{}, apperr.ErrUnauthorized
	}
	if err != nil {
		return store.User{}, err
	}
	if sess.UserID != a.UserID || sess.RevokedAt != nil || !s.d.Clock.Now().Before(sess.ExpiresAt) {
		return store.User{}, apperr.ErrUnauthorized
	}
	return s.Me(ctx, a)
}

// checkPassword verifies the current password. The attempt is reserved before the check, so
// parallel guesses can't all slip past the limit; only a wrong password keeps it spent.
func (s *Service) checkPassword(ctx context.Context, u store.User, pw string) error {
	refund := func() {}
	if l := s.d.PasswordFailures; l != nil {
		r, allowed, wait := l.Acquire(u.ID.String())
		if !allowed {
			return apperr.RateLimited(wait)
		}
		refund = r
	}
	ok, _, err := s.d.Hasher.Verify(ctx, pw, u.PasswordHash)
	if err != nil {
		refund()
		return err
	}
	if !ok {
		return apperr.ErrWrongPassword
	}
	refund()
	return nil
}

// ChangePassword keeps the current session and revokes all others.
func (s *Service) ChangePassword(ctx context.Context, a token.Access, oldPassword, newPassword string) error {
	if fe := password.FieldError("new_password", newPassword); fe != nil {
		return apperr.Validation(*fe)
	}
	u, err := s.activeUser(ctx, a)
	if err != nil {
		return err
	}
	if err := s.checkPassword(ctx, u, oldPassword); err != nil {
		return err
	}
	hash, err := s.d.Hasher.Hash(ctx, newPassword)
	if err != nil {
		return err
	}
	now := s.d.Clock.Now()
	return s.d.Store.InTx(ctx, func(q *store.Queries) error {
		if err := q.UpdatePasswordHash(ctx, store.UpdatePasswordHashParams{ID: u.ID, PasswordHash: hash, Now: now}); err != nil {
			return err
		}
		return q.RevokeUserSessionsExcept(ctx, store.RevokeUserSessionsExceptParams{
			UserID: u.ID, KeepID: a.SessionID, Now: now, Reason: "password_change",
		})
	})
}

// Delete removes the user; sessions and codes go with it (ON DELETE CASCADE).
func (s *Service) Delete(ctx context.Context, a token.Access, pw string) error {
	u, err := s.activeUser(ctx, a)
	if err != nil {
		return err
	}
	if err := s.checkPassword(ctx, u, pw); err != nil {
		return err
	}
	return s.d.Store.DeleteUser(ctx, u.ID)
}
```

```go
package account

import (
	"context"

	"github.com/AntiqueViolet/go_kachalka_backend/internal/apperr"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/contact"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/otp"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/store"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/token"
)

// ChangeEmail replaces the email (unverified until confirmed) and sends a code to it.
// The password is required: otherwise a stolen access token could redirect password resets.
func (s *Service) ChangeEmail(ctx context.Context, a token.Access, rawEmail, pw string) (store.User, CodeSent, error) {
	email, err := contact.NormalizeEmail(rawEmail)
	if err != nil {
		return store.User{}, CodeSent{}, apperr.Validation(apperr.FieldError{Field: "email", Code: apperr.FieldInvalidEmail})
	}
	u, err := s.activeUser(ctx, a)
	if err != nil {
		return store.User{}, CodeSent{}, err
	}
	if err := s.checkPassword(ctx, u, pw); err != nil {
		return store.User{}, CodeSent{}, err
	}
	if email == u.Email {
		if u.EmailVerifiedAt != nil {
			return store.User{}, CodeSent{}, apperr.ErrEmailAlreadyVerified
		}
		if err := s.d.Codes.SendEmailCode(ctx, otp.PurposeEmailVerify, u.ID, u.Email); err != nil {
			return store.User{}, CodeSent{}, err
		}
		return u, s.codeSent(), nil
	}

	var issued otp.Issued
	err = s.d.Store.InTx(ctx, func(q *store.Queries) error {
		var err error
		if u, err = q.SetEmail(ctx, store.SetEmailParams{ID: u.ID, Email: email, Now: s.d.Clock.Now()}); err != nil {
			return err
		}
		// Issued in the same transaction: if the address is rate limited the change is rolled back.
		issued, err = s.d.OTP.Issue(ctx, q, otp.PurposeEmailVerify, email, &u.ID)
		return err
	})
	if err != nil {
		return store.User{}, CodeSent{}, err
	}
	// Delivery failure is logged by Codes; the user can ask for a resend.
	_ = s.d.Codes.DeliverEmailCode(ctx, otp.PurposeEmailVerify, email, issued.Code)
	return u, s.codeSent(), nil
}

func (s *Service) VerifyEmail(ctx context.Context, a token.Access, code string) (store.User, error) {
	u, err := s.Me(ctx, a)
	if err != nil {
		return store.User{}, err
	}
	if u.EmailVerifiedAt != nil {
		return store.User{}, apperr.ErrEmailAlreadyVerified
	}
	c, err := s.d.Store.LatestActiveUserCode(ctx, store.LatestActiveUserCodeParams{Purpose: string(otp.PurposeEmailVerify), UserID: u.ID})
	if store.IsNotFound(err) || (err == nil && c.Target != u.Email) {
		return store.User{}, apperr.ErrCodeExpired
	}
	if err != nil {
		return store.User{}, err
	}
	if err := s.d.OTP.Check(ctx, s.d.Store.Queries, c, code); err != nil {
		return store.User{}, err
	}
	// User row first, then the code row: the same order as ChangeEmail, so the two can't deadlock.
	err = s.d.Store.InTx(ctx, func(q *store.Queries) error {
		var err error
		u, err = q.MarkEmailVerified(ctx, store.MarkEmailVerifiedParams{ID: u.ID, Now: s.d.Clock.Now()})
		if store.IsUniqueViolation(err, store.ConstraintVerifiedEmail) {
			return apperr.ErrEmailTaken
		}
		if err != nil {
			return err
		}
		// The email may have changed since the code was looked up.
		if u.Email != c.Target {
			return apperr.ErrCodeExpired
		}
		return s.d.OTP.Consume(ctx, q, c.ID)
	})
	if err != nil {
		return store.User{}, err
	}
	return u, nil
}

func (s *Service) ResendEmailCode(ctx context.Context, a token.Access) (CodeSent, error) {
	u, err := s.Me(ctx, a)
	if err != nil {
		return CodeSent{}, err
	}
	if u.EmailVerifiedAt != nil {
		return CodeSent{}, apperr.ErrEmailAlreadyVerified
	}
	if err := s.d.Codes.SendEmailCode(ctx, otp.PurposeEmailVerify, u.ID, u.Email); err != nil {
		return CodeSent{}, err
	}
	return s.codeSent(), nil
}
```

```go
func TestClientIP(t *testing.T) {
	trusted := []netip.Prefix{netip.MustParsePrefix("172.18.0.0/16")}
	cases := []struct {
		name   string
		remote string
		xff    string
		want   string
	}{
		{"direct client", "203.0.113.7:5000", "", "203.0.113.7"},
		{"untrusted peer cannot spoof", "203.0.113.7:5000", "1.2.3.4", "203.0.113.7"},
		{"trusted proxy", "172.18.0.2:5000", "198.51.100.9", "198.51.100.9"},
		{"client-supplied prefix ignored", "172.18.0.2:5000", "1.2.3.4, 198.51.100.9", "198.51.100.9"},
		{"chain of trusted proxies", "172.18.0.2:5000", "198.51.100.9, 172.18.0.3", "198.51.100.9"},
		{"trusted proxy without header", "172.18.0.2:5000", "", "172.18.0.2"},
		{"ipv4-mapped ipv6", "[::ffff:203.0.113.7]:5000", "", "203.0.113.7"},
	}
	for _, c := range cases {
		r := httptest.NewRequest("GET", "/", nil)
		r.RemoteAddr = c.remote
		if c.xff != "" {
			r.Header.Set("X-Forwarded-For", c.xff)
		}
		if got := ClientIP(r, trusted).String(); got != c.want {
			t.Errorf("%s: got %s, want %s", c.name, got, c.want)
		}
	}
}
```

```go
// writeProblem renders e as RFC 9457 problem details.
func writeProblem(w http.ResponseWriter, e *apperr.Error) {
	h := w.Header()
	if e.RetryAfter > 0 {
		h.Set("Retry-After", strconv.Itoa(int(math.Ceil(e.RetryAfter.Seconds()))))
	}
	if e.Status == http.StatusUnauthorized {
		h.Set("WWW-Authenticate", "Bearer")
	}
	h.Set("Content-Type", "application/problem+json")
	w.WriteHeader(e.Status)
	_ = json.NewEncoder(w).Encode(toProblem(e))
}

func toProblem(e *apperr.Error) oapi.Problem {
	p := oapi.Problem{
		Type:         "about:blank",
		Title:        e.Title,
		Status:       e.Status,
		Code:         oapi.ProblemCode(e.Code),
		AttemptsLeft: e.AttemptsLeft,
	}
	if len(e.Fields) > 0 {
		fields := make([]oapi.FieldError, len(e.Fields))
		for i, f := range e.Fields {
			fields[i] = oapi.FieldError{Field: f.Field, Code: oapi.FieldErrorCode(f.Code)}
		}
		p.Errors = &fields
	}
	return p
}

// errorHandlers plug into the generated strict server.
type errorHandlers struct{ log *slog.Logger }

// request handles bodies the generated code could not decode (bad JSON, too large).
func (h errorHandlers) request(w http.ResponseWriter, r *http.Request, err error) {
	h.log.DebugContext(r.Context(), "bad request body", "err", err)
	writeProblem(w, apperr.ErrBadRequest)
}

// response maps errors returned by handlers and middlewares. Anything that is not an
// *apperr.Error is a bug or an outage: it is logged and hidden behind internal_error.
func (h errorHandlers) response(w http.ResponseWriter, r *http.Request, err error) {
	var ae *apperr.Error
	switch {
	case errors.As(err, &ae):
	case errors.Is(err, password.ErrBusy):
		ae = apperr.ErrUnavailable
	case errors.Is(err, context.DeadlineExceeded):
		// requestTimeout fired, e.g. while waiting for a hashing slot.
		h.log.WarnContext(r.Context(), "request timed out", "err", err, "request_id", RequestID(r.Context()))
		ae = apperr.ErrUnavailable
	case errors.Is(err, context.Canceled):
		// The client went away; nobody reads the response, and it is not a server fault.
		h.log.DebugContext(r.Context(), "request canceled", "err", err, "request_id", RequestID(r.Context()))
		ae = apperr.ErrUnavailable
	default:
		h.log.ErrorContext(r.Context(), "request failed", "err", err, "request_id", RequestID(r.Context()))
		ae = apperr.ErrInternal
	}
	writeProblem(w, ae)
}
```

```go
package httpapi

import (
	"net/http"
	"net/netip"
	"strings"
)

// ClientIP returns the caller's address. X-Forwarded-For is honoured only when the direct
// peer is a trusted proxy; then the right-most hop that is not itself trusted is the client.
// Without this check anyone could spoof the header and dodge per-IP limits.
func ClientIP(r *http.Request, trusted []netip.Prefix) netip.Addr {
	peer, err := netip.ParseAddrPort(r.RemoteAddr)
	if err != nil {
		return netip.Addr{}
	}
	addr := peer.Addr().Unmap()
	if !isTrusted(addr, trusted) {
		return addr
	}
	hops := strings.Split(strings.Join(r.Header.Values("X-Forwarded-For"), ","), ",")
	for i := len(hops) - 1; i >= 0; i-- {
		hop, err := netip.ParseAddr(strings.TrimSpace(hops[i]))
		if err != nil {
			break
		}
		hop = hop.Unmap()
		if !isTrusted(hop, trusted) {
			return hop
		}
		addr = hop
	}
	return addr
}

func isTrusted(a netip.Addr, trusted []netip.Prefix) bool {
	for _, p := range trusted {
		if p.Contains(a) {
			return true
		}
	}
	return false
}
```

```go
type requestIDKey struct{}

var validRequestID = regexp.MustCompile(`^[A-Za-z0-9._-]{1,64}$`)

// withRequestID keeps a sane incoming X-Request-ID or makes one, and echoes it back.
func withRequestID(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		id := r.Header.Get("X-Request-ID")
		if !validRequestID.MatchString(id) {
			id = rand.Text()[:16]
		}
		w.Header().Set("X-Request-ID", id)
		next.ServeHTTP(w, r.WithContext(context.WithValue(r.Context(), requestIDKey{}, id)))
	})
}

// RequestID returns the id assigned by withRequestID.
func RequestID(ctx context.Context) string {
	id, _ := ctx.Value(requestIDKey{}).(string)
	return id
}

type statusWriter struct {
	http.ResponseWriter
	status int
}

func (w *statusWriter) WriteHeader(code int) {
	if w.status == 0 {
		w.status = code
	}
	w.ResponseWriter.WriteHeader(code)
}

func (w *statusWriter) Write(b []byte) (int, error) {
	if w.status == 0 {
		w.status = http.StatusOK
	}
	return w.ResponseWriter.Write(b)
}

// withAccessLog logs one line per request. Bodies are never logged: they hold passwords and codes.
func withAccessLog(log *slog.Logger, ipOf func(*http.Request) netip.Addr, next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		start := time.Now()
		sw := &statusWriter{ResponseWriter: w}
		next.ServeHTTP(sw, r)
		level := slog.LevelInfo
		if strings.HasPrefix(r.URL.Path, "/healthz") || strings.HasPrefix(r.URL.Path, "/readyz") {
			level = slog.LevelDebug
		}
		log.Log(r.Context(), level, "http request",
			"method", r.Method,
			"path", r.URL.Path,
			"status", sw.status,
			"duration_ms", time.Since(start).Milliseconds(),
			"ip", ipOf(r).String(),
			"request_id", RequestID(r.Context()),
		)
	})
}

// withRecover turns a panic into a logged 500 instead of a dropped connection.
func withRecover(log *slog.Logger, next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		defer func() {
			if v := recover(); v != nil {
				if err, ok := v.(error); ok && errors.Is(err, http.ErrAbortHandler) {
					panic(v) // net/http's deliberate abort; let the server handle it
				}
				log.ErrorContext(r.Context(), "panic", "value", v, "stack", string(debug.Stack()), "request_id", RequestID(r.Context()))
				writeProblem(w, apperr.ErrInternal)
			}
		}()
		next.ServeHTTP(w, r)
	})
}

// requestTimeout bounds one request. http.Server.WriteTimeout does not cancel r.Context(), so
// without it a request waiting for a password-hashing slot would wait forever and the designed
// 503 would never fire. Below the server's WriteTimeout (15 s), above the SMTP send limit (10 s).
// A variable only so tests can shorten it.
var requestTimeout = 12 * time.Second

// withLimits caps the request body and the request time, and forbids caching of API responses
// (they carry tokens).
func withLimits(maxBody int64, next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		ctx, cancel := context.WithTimeout(r.Context(), requestTimeout)
		defer cancel()
		r = r.WithContext(ctx)
		r.Body = http.MaxBytesReader(w, r.Body, maxBody)
		w.Header().Set("Cache-Control", "no-store")
		next.ServeHTTP(w, r)
	})
}

// ---- strict middlewares (per operation; operationID as in the generated code) ----

// protectedOps lists operations with `security: bearerAuth` in api/openapi.yaml.
var protectedOps = map[string]bool{
	"Logout":          true,
	"GetMe":           true,
	"DeleteMe":        true,
	"ChangeEmail":     true,
	"VerifyEmail":     true,
	"ResendEmailCode": true,
	"ChangePassword":  true,
}

type principalKey struct{}

func withPrincipal(ctx context.Context, a token.Access) context.Context {
	return context.WithValue(ctx, principalKey{}, a)
}

// principal returns the caller of a protected operation.
func principal(ctx context.Context) (token.Access, error) {
	a, ok := ctx.Value(principalKey{}).(token.Access)
	if !ok {
		return token.Access{}, apperr.ErrUnauthorized
	}
	return a, nil
}

// requireAuth validates the bearer access token for protected operations.
func requireAuth(signer *token.Signer) oapi.StrictMiddlewareFunc {
	return func(f oapi.StrictHandlerFunc, op string) oapi.StrictHandlerFunc {
		if !protectedOps[op] {
			return f
		}
		return func(ctx context.Context, w http.ResponseWriter, r *http.Request, req any) (any, error) {
			// The auth scheme is case-insensitive (RFC 7235).
			const prefix = "Bearer "
			h := r.Header.Get("Authorization")
			if len(h) <= len(prefix) || !strings.EqualFold(h[:len(prefix)], prefix) {
				return nil, apperr.ErrUnauthorized
			}
			raw := h[len(prefix):]
			a, err := signer.ParseAccess(raw)
			if err != nil {
				return nil, apperr.ErrUnauthorized
			}
			return f(withPrincipal(ctx, a), w, r, req)
		}
	}
}

// limitKey groups IPv6 clients by /64: one host usually controls a whole /64, so per-address
// buckets would let it multiply its limit (and the limiter's memory).
func limitKey(a netip.Addr) string {
	if a.Is6() && !a.Is4In6() {
		if p, err := a.Prefix(64); err == nil {
			return p.String()
		}
	}
	return a.String()
}

// limitByIP applies a per-client-IP limiter to the listed operations.
func limitByIP(limits map[string]*ratelimit.Limiter, ipOf func(*http.Request) netip.Addr) oapi.StrictMiddlewareFunc {
	return func(f oapi.StrictHandlerFunc, op string) oapi.StrictHandlerFunc {
		lim, ok := limits[op]
		if !ok {
			return f
		}
		return func(ctx context.Context, w http.ResponseWriter, r *http.Request, req any) (any, error) {
			if ok, wait := lim.Allow(limitKey(ipOf(r))); !ok {
				return nil, apperr.RateLimited(wait)
			}
			return f(ctx, w, r, req)
		}
	}
}
```

```go
// Package httpapi exposes the services over HTTP according to api/openapi.yaml.
package httpapi

import (
	"context"
	"encoding/json"
	"log/slog"
	"net/http"
	"net/netip"
	"time"

	"github.com/AntiqueViolet/go_kachalka_backend/internal/account"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/apperr"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/auth"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/httpapi/oapi"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/ratelimit"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/token"
)

const maxBodyBytes = 64 << 10

type Pinger interface {
	Ping(ctx context.Context) error
}

// Limits are per-client-IP limiters (spec section 8).
type Limits struct {
	Login   *ratelimit.Limiter
	Refresh *ratelimit.Limiter
	Codes   *ratelimit.Limiter // shared by every operation that sends a code
}

type Config struct {
	Auth           *auth.Service
	Account        *account.Service
	Signer         *token.Signer
	DB             Pinger
	Log            *slog.Logger
	TrustedProxies []netip.Prefix
	Limits         Limits
}

func New(c Config) http.Handler {
	ipOf := func(r *http.Request) netip.Addr { return ClientIP(r, c.TrustedProxies) }
	errs := errorHandlers{log: c.Log}

	mux := http.NewServeMux()
	mux.HandleFunc("GET /healthz", func(w http.ResponseWriter, _ *http.Request) {
		writeJSON(w, http.StatusOK, map[string]string{"status": "ok"})
	})
	mux.HandleFunc("GET /readyz", func(w http.ResponseWriter, r *http.Request) {
		ctx, cancel := context.WithTimeout(r.Context(), 2*time.Second)
		defer cancel()
		if err := c.DB.Ping(ctx); err != nil {
			c.Log.WarnContext(r.Context(), "readiness check failed", "err", err, "request_id", RequestID(r.Context()))
			writeProblem(w, apperr.ErrUnavailable)
			return
		}
		writeJSON(w, http.StatusOK, map[string]string{"status": "ready"})
	})

	strict := oapi.NewStrictHandlerWithOptions(
		handlers{auth: c.Auth, account: c.Account},
		// The last middleware is the outermost: authentication runs before the per-IP limits,
		// so anonymous (401) calls to protected operations never spend the shared quota.
		[]oapi.StrictMiddlewareFunc{
			limitByIP(ipLimits(c.Limits), ipOf),
			requireAuth(c.Signer),
		},
		oapi.StrictHTTPServerOptions{
			RequestErrorHandlerFunc:  errs.request,
			ResponseErrorHandlerFunc: errs.response,
		},
	)
	oapi.HandlerWithOptions(strict, oapi.StdHTTPServerOptions{BaseRouter: mux})

	var h http.Handler = mux
	h = withLimits(maxBodyBytes, h)
	h = withRecover(c.Log, h)
	h = withAccessLog(c.Log, ipOf, h)
	h = withRequestID(h)
	return h
}

// ipLimits maps operations (names as in the generated strict handler) to their per-IP limiter.
func ipLimits(l Limits) map[string]*ratelimit.Limiter {
	return map[string]*ratelimit.Limiter{
		"Login":           l.Login,
		"Refresh":         l.Refresh,
		"RegisterStart":   l.Codes,
		"ForgotPassword":  l.Codes,
		"ChangeEmail":     l.Codes,
		"ResendEmailCode": l.Codes,
	}
}

func writeJSON(w http.ResponseWriter, status int, v any) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)
	_ = json.NewEncoder(w).Encode(v)
}
```

```go
package app

import (
	"context"
	"crypto/ed25519"
	"log/slog"
	"net/http"
	"time"

	"github.com/jackc/pgx/v5/pgxpool"

	"github.com/AntiqueViolet/go_kachalka_backend/internal/account"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/auth"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/clock"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/codes"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/config"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/httpapi"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/notify"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/otp"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/password"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/ratelimit"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/store"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/token"
)

const (
	registrationTTL = 15 * time.Minute
	keepDeadSession = 30 * 24 * time.Hour
	keepCodes       = 24 * time.Hour
)

type Options struct {
	Config config.Config
	Pool   *pgxpool.Pool
	Log    *slog.Logger

	// Optional overrides for tests; zero values mean "build from Config".
	Clock          clock.Clock
	SMS            notify.SMSSender
	Email          notify.EmailSender
	JWTKey         ed25519.PrivateKey
	PasswordParams *password.Params
}

type App struct {
	Handler http.Handler
	store   *store.Store
	clock   clock.Clock
	log     *slog.Logger
	codes   *codes.Sender
}

func New(o Options) (*App, error) {
	cfg := o.Config
	clk := o.Clock
	if clk == nil {
		clk = clock.Real{}
	}
	key, err := jwtKey(o)
	if err != nil {
		return nil, err
	}
	params := password.DefaultParams
	if o.PasswordParams != nil {
		params = *o.PasswordParams
	}
	hasher, err := password.NewHasher(params, cfg.PasswordHashConcurrency)
	if err != nil {
		return nil, err
	}
	sms, email, err := senders(o)
	if err != nil {
		return nil, err
	}

	st := store.NewStore(o.Pool)
	signer := token.NewSigner(token.SignerConfig{
		Key: key, Issuer: cfg.JWTIssuer, AccessTTL: cfg.AccessTokenTTL, RegistrationTTL: registrationTTL, Clock: clk,
	})
	otpSvc := otp.NewService(cfg.Keys.OTP, otp.Limits{Cooldown: cfg.Rate.CodeCooldown, PerTargetPerHour: cfg.Rate.CodesPerTargetPerHour}, clk)
	sender := codes.NewSender(codes.Deps{
		Store: st, OTP: otpSvc, SMS: sms, Email: email, Log: o.Log,
		Texts: notify.Texts{AppName: cfg.AppName, SMSAppHash: cfg.SMSAppHash, CodeTTL: otp.TTL},
	})
	authSvc := auth.New(auth.Deps{
		Store: st, Hasher: hasher, Signer: signer, Refresh: token.NewRefreshCodec(cfg.Keys.Refresh),
		OTP: otpSvc, Codes: sender, Clock: clk, Log: o.Log,
		LoginFailures: ratelimit.Per(cfg.Rate.LoginFailuresPerPhone, cfg.Rate.LoginFailuresWindow, clk),
		RefreshTTL:    cfg.RefreshTokenTTL,
		RefreshGrace:  cfg.RefreshGrace,
	})
	accountSvc := account.New(account.Deps{
		Store: st, Hasher: hasher, OTP: otpSvc, Codes: sender, Clock: clk, Log: o.Log,
		// Its own limiter: sharing auth's would mix the two flows' failure counts.
		PasswordFailures: ratelimit.Per(cfg.Rate.LoginFailuresPerPhone, cfg.Rate.LoginFailuresWindow, clk),
	})

	h := httpapi.New(httpapi.Config{
		Auth: authSvc, Account: accountSvc, Signer: signer, DB: st, Log: o.Log,
		TrustedProxies: cfg.ProxyPrefixes,
		Limits: httpapi.Limits{
			Login:   ratelimit.Per(cfg.Rate.LoginPerIPPerMinute, time.Minute, clk),
			Refresh: ratelimit.Per(cfg.Rate.RefreshPerIPPerMinute, time.Minute, clk),
			Codes:   ratelimit.Per(cfg.Rate.CodesPerIPPerHour, time.Hour, clk),
		},
	})
	return &App{Handler: h, store: st, clock: clk, log: o.Log, codes: sender}, nil
}

func jwtKey(o Options) (ed25519.PrivateKey, error) {
	switch {
	case o.JWTKey != nil:
		return o.JWTKey, nil
	case o.Config.JWTPrivateKeyFile != "":
		return token.LoadPrivateKey(o.Config.JWTPrivateKeyFile)
	default: // dev only: config requires the key file in prod
		o.Log.Warn("JWT_PRIVATE_KEY_FILE is empty: using a temporary key, tokens die on restart")
		return token.GenerateKey()
	}
}

func senders(o Options) (notify.SMSSender, notify.EmailSender, error) {
	sms, email := o.SMS, o.Email
	if sms == nil {
		// config.validate guarantees "log" is the only provider, and only outside prod.
		sms = notify.NewLogSMS(o.Log)
	}
	if email == nil {
		if o.Config.SMTP.Host == "" {
			o.Log.Warn("SMTP_HOST is empty: emails are written to the log")
			email = notify.NewLogEmail(o.Log)
		} else {
			c := o.Config.SMTP
			smtp, err := notify.NewSMTP(notify.SMTPConfig{
				Host: c.Host, Port: c.Port, Username: c.Username, Password: c.Password, From: c.From, TLS: c.TLS,
			})
			if err != nil {
				return nil, nil, err
			}
			email = smtp
		}
	}
	return sms, email, nil
}

// Cleanup deletes sessions dead for over 30 days and codes older than a day.
func (a *App) Cleanup(ctx context.Context) error {
	now := a.clock.Now()
	sessions, err := a.store.DeleteStaleSessions(ctx, now.Add(-keepDeadSession))
	if err != nil {
		return err
	}
	codesDeleted, err := a.store.DeleteOldCodes(ctx, now.Add(-keepCodes))
	if err != nil {
		return err
	}
	a.log.InfoContext(ctx, "cleanup done", "sessions", sessions, "codes", codesDeleted)
	return nil
}

// Wait blocks until background email sends have finished; call it after the HTTP
// server stops so shutdown does not drop in-flight messages.
func (a *App) Wait() { a.codes.Wait() }

// RunCleanup calls Cleanup at once and then every interval until ctx is cancelled.
// The immediate pass matters: a service redeployed more often than the interval
// would otherwise never clean anything.
func (a *App) RunCleanup(ctx context.Context, every time.Duration) {
	t := time.NewTicker(every)
	defer t.Stop()
	for {
		if err := a.Cleanup(ctx); err != nil && ctx.Err() == nil {
			a.log.ErrorContext(ctx, "cleanup failed", "err", err)
		}
		select {
		case <-ctx.Done():
			return
		case <-t.C:
		}
	}
}
```

### Task 11: Бинарники `api` и `keygen`

`api` ждёт БД до 30 секунд (под docker compose она может ещё стартовать), применяет миграции и только потом открывает порт (до этого клиенты и healthcheck получают отказ в соединении, а не зависают), обслуживает запросы и корректно завершается по SIGTERM: дожидается запросов (до 20 с), фоновых писем и очистки. Сигнал во время старта — штатная остановка, не ошибка. `api healthcheck` нужен для `HEALTHCHECK` в distroless-образе, где нет curl. `keygen` никогда не перезаписывает существующий ключ, создаёт каталог для него и не оставляет недописанный файл.

```go
func TestRunServesAndShutsDown(t *testing.T) {
	cfg, err := config.LoadFrom(map[string]string{"DATABASE_URL": storetest.DSN(t)})
	if err != nil {
		t.Fatal(err)
	}
	// run binds the port itself, after the database is ready; learn the address from it.
	addrc := make(chan string, 1)
	listen := func() (net.Listener, error) {
		ln, err := net.Listen("tcp", "127.0.0.1:0")
		if err == nil {
			addrc <- ln.Addr().String()
		}
		return ln, err
	}
	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()
	done := make(chan error, 1)
	go func() { done <- run(ctx, cfg, slog.New(slog.NewTextHandler(io.Discard, nil)), listen) }()

	var addr string
	select {
	case addr = <-addrc:
	case err := <-done:
		t.Fatalf("run returned before listening: %v", err)
	case <-time.After(30 * time.Second):
		t.Fatal("run never listened")
	}
	base := "http://" + addr
	deadline := time.Now().Add(15 * time.Second)
	for {
		resp, err := http.Get(base + "/readyz")
		if err == nil {
			resp.Body.Close()
			if resp.StatusCode == http.StatusOK {
				break
			}
		}
		if time.Now().After(deadline) {
			t.Fatalf("server not ready: %v", err)
		}
		time.Sleep(50 * time.Millisecond)
	}
	if err := healthcheck(addr); err != nil {
		t.Fatalf("healthcheck: %v", err)
	}

	cancel()
	select {
	case err := <-done:
		if err != nil {
			t.Fatalf("run returned %v", err)
		}
	case <-time.After(shutdownTimeout + 5*time.Second):
		t.Fatal("server did not shut down")
	}
}

func TestHealthcheckFailsWithoutServer(t *testing.T) {
	ln, err := net.Listen("tcp", "127.0.0.1:0")
	if err != nil {
		t.Fatal(err)
	}
	addr := ln.Addr().String()
	ln.Close()
	if err := healthcheck(addr); err == nil {
		t.Fatal("healthcheck succeeded with nothing listening")
	}
}
```

```go
	}
```

```go
// Command api is the go_kachalka backend server.
//
//	api              serve HTTP (configuration from environment variables, see README)
//	api healthcheck  exit 0 if the local server answers /healthz (for Docker HEALTHCHECK)
package main

import (
	"context"
	"fmt"
	"log/slog"
	"net"
	"os"
	"os/signal"
	"syscall"

	"github.com/AntiqueViolet/go_kachalka_backend/internal/config"
)

func main() {
	var err error
	if len(os.Args) > 1 && os.Args[1] == "healthcheck" {
		addr := os.Getenv("HTTP_ADDR")
		if addr == "" {
			addr = ":8080"
		}
		err = healthcheck(addr)
	} else {
		err = serve()
	}
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
}

func serve() error {
	cfg, err := config.Load()
	if err != nil {
		return fmt.Errorf("config: %w", err)
	}
	log := slog.New(slog.NewJSONHandler(os.Stdout, &slog.HandlerOptions{Level: cfg.Level}))
	ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
	defer stop()
	// run binds the port only once the database is ready, see run.
	listen := func() (net.Listener, error) {
		return (&net.ListenConfig{}).Listen(ctx, "tcp", cfg.HTTPAddr)
	}
	return run(ctx, cfg, log, listen)
}
```

```go
package main

import (
	"context"
	"errors"
	"fmt"
	"log/slog"
	"net"
	"net/http"
	"time"

	"github.com/jackc/pgx/v5/pgxpool"

	"github.com/AntiqueViolet/go_kachalka_backend/internal/app"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/config"
	"github.com/AntiqueViolet/go_kachalka_backend/internal/store"
)

const (
	shutdownTimeout = 20 * time.Second
	cleanupEvery    = time.Hour
	dbWaitAttempts  = 30
)

// run migrates the database, then binds the port via listen and serves until ctx is
// cancelled, then drains in-flight requests for up to shutdownTimeout. The port is bound
// only after the database is ready, so clients and the Docker healthcheck get
// "connection refused" instead of hanging while the service is not usable yet.
// A signal during startup is a clean stop, not an error.
func run(ctx context.Context, cfg config.Config, log *slog.Logger, listen func() (net.Listener, error)) error {
	if cfg.UsingDevSecret() {
		log.Warn("APP_SECRET is empty: using the insecure dev secret")
	}
	pool, err := pgxpool.New(ctx, cfg.DatabaseURL)
	if err != nil {
		return fmt.Errorf("database: %w", err)
	}
	defer pool.Close()
	if err := waitForDB(ctx, pool, log); err != nil {
		return startupErr(ctx, log, err)
	}
	if err := store.Migrate(ctx, pool); err != nil {
		return startupErr(ctx, log, err)
	}
	a, err := app.New(app.Options{Config: cfg, Pool: pool, Log: log})
	if err != nil {
		return err
	}
	ln, err := listen()
	if err != nil {
		return err
	}

	srv := &http.Server{
		Handler:           a.Handler,
		ReadHeaderTimeout: 5 * time.Second,
		ReadTimeout:       10 * time.Second,
		WriteTimeout:      15 * time.Second,
		IdleTimeout:       60 * time.Second,
		ErrorLog:          slog.NewLogLogger(log.Handler(), slog.LevelWarn),
	}
	cleanupCtx, stopCleanup := context.WithCancel(ctx)
	defer stopCleanup()
	cleanupDone := make(chan struct{})
	go func() {
		defer close(cleanupDone)
		a.RunCleanup(cleanupCtx, cleanupEvery)
	}()

	errc := make(chan error, 1)
	go func() { errc <- srv.Serve(ln) }()
	log.Info("listening", "addr", ln.Addr().String(), "env", cfg.Env)

	select {
	case err := <-errc:
		stopCleanup()
		<-cleanupDone
		a.Wait()
		return err
	case <-ctx.Done():
	}
	log.Info("shutting down")
	sctx, cancel := context.WithTimeout(context.Background(), shutdownTimeout)
	defer cancel()
	if err := srv.Shutdown(sctx); err != nil {
		_ = srv.Close() // drain timed out: drop the remaining connections
		return err
	}
	if err := <-errc; !errors.Is(err, http.ErrServerClosed) {
		return err
	}
	// The server is stopped: let the cleanup pass and background emails finish
	// before the deferred pool.Close pulls the database away.
	<-cleanupDone
	a.Wait()
	return nil
}

// startupErr turns the error caused by ctx being cancelled during startup into nil.
func startupErr(ctx context.Context, log *slog.Logger, err error) error {
	if ctx.Err() != nil {
		log.Info("startup interrupted")
		return nil
	}
	return err
}

// waitForDB retries the first connection for up to dbWaitAttempts seconds: under docker
// compose the database may still be starting.
func waitForDB(ctx context.Context, pool *pgxpool.Pool, log *slog.Logger) error {
	var err error
	for attempt := 1; attempt <= dbWaitAttempts; attempt++ {
		if err = pool.Ping(ctx); err == nil {
			return nil
		}
		log.Warn("database not ready, retrying", "attempt", attempt, "err", err)
		select {
		case <-ctx.Done():
			return ctx.Err()
		case <-time.After(time.Second):
		}
	}
	return fmt.Errorf("database: %w", err)
}

// healthcheck asks the server listening on addr (HTTP_ADDR) for /healthz.
func healthcheck(addr string) error {
	host, port, err := net.SplitHostPort(addr)
	if err != nil {
		return err
	}
	if host == "" || host == "0.0.0.0" || host == "::" {
		host = "127.0.0.1"
	}
	ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
	defer cancel()
	// gosec G704 (SSRF) does not apply: the URL is built from our own HTTP_ADDR.
	url := "http://" + net.JoinHostPort(host, port) + "/healthz"
	req, err := http.NewRequestWithContext(ctx, http.MethodGet, url, nil) //nolint:gosec
	if err != nil {
		return err
	}
	resp, err := http.DefaultClient.Do(req) //nolint:gosec
	if err != nil {
		return err
	}
	_ = resp.Body.Close()
	if resp.StatusCode != http.StatusOK {
		return fmt.Errorf("healthz: status %d", resp.StatusCode)
	}
	return nil
}
```

```go
// Command keygen creates the production secrets: an Ed25519 private key for signing JWTs
// (written to a file) and APP_SECRET (printed to stdout).
//
//	keygen -jwt-key secrets/jwt.pem
package main

import (
	"crypto/rand"
	"encoding/base64"
	"flag"
	"fmt"
	"io"
	"os"
	"path/filepath"

	"github.com/AntiqueViolet/go_kachalka_backend/internal/token"
)

func main() {
	keyPath := flag.String("jwt-key", "jwt.pem", "where to write the JWT private key; the file must not exist")
	flag.Parse()
	if err := generate(*keyPath, os.Stdout); err != nil {
		fmt.Fprintln(os.Stderr, "keygen:", err)
		os.Exit(1)
	}
}

func generate(keyPath string, out io.Writer) error {
	key, err := token.GenerateKey()
	if err != nil {
		return err
	}
	pemBytes, err := token.EncodePrivateKeyPEM(key)
	if err != nil {
		return err
	}
	// Draw the secret before creating the file: a rand failure must not leave a key
	// behind that O_EXCL then refuses to overwrite on the next run.
	secret := make([]byte, 32)
	if _, err := rand.Read(secret); err != nil {
		return err
	}
	if err := writeNew(keyPath, pemBytes); err != nil {
		return err
	}
	_, err = fmt.Fprintf(out, "APP_SECRET=%s\n", base64.StdEncoding.EncodeToString(secret))
	return err
}

// writeNew creates path (0600, parent directory 0700) and removes it again if the
// write does not complete, so a half-written key never blocks the next run.
func writeNew(path string, data []byte) error {
	if err := os.MkdirAll(filepath.Dir(path), 0o700); err != nil {
		return err
	}
	// O_EXCL: never overwrite an existing key, that would log every user out.
	f, err := os.OpenFile(path, os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0o600)
	if err != nil {
		return err
	}
	_, err = f.Write(data)
	if err == nil {
		err = f.Sync()
	}
	if cerr := f.Close(); err == nil {
		err = cerr
	}
	if err != nil {
		_ = os.Remove(path)
	}
	return err
}
```
